    private Asignatura asig(String codigo, String nombre, int creditos, ProgramaAcademico programa) {
        return asignaturas.save(new Asignatura(codigo, nombre, creditos, programa));
    }

    private void curso(Asignatura a, PeriodoAcademico p, String grupo, int cupo, String docente, FranjaHoraria... franjas) {
        cursos.save(new Curso(a, p, grupo, cupo, docente, List.of(franjas)));
    }

    private static FranjaHoraria f(DiaSemana dia, int desde, int hasta, String aula) {
        return new FranjaHoraria(dia, LocalTime.of(desde, 0), LocalTime.of(hasta, 0), aula);
    }

    private void estudiante(String usuario, String nombre, String codigo, ProgramaAcademico programa,
                            EstadoAcademico estado, String hash) {
        String correo = usuario + "@estudiantes.uta.edu.co";
        usuarios.save(new Estudiante(usuario, nombre, correo, hash, codigo, programa, estado));
    }

    private void confirmar(String usuario, List<Curso> todos, String... pares) {
        List<Long> ids = new ArrayList<>();
        for (int i = 0; i < pares.length; i += 2) {
            String nombre = pares[i];
            String grupo = pares[i + 1];
            todos.stream().filter(c -> c.getAsignatura().getNombre().equals(nombre) && c.getGrupo().equals(grupo))
                    .findFirst().ifPresent(c -> ids.add(c.getId()));
        }
        Matricula m = matriculaService.registrarMatricula(usuario, ids);
        pasarela.responder(m.getPago().getReferencia(), true);
        matriculaService.confirmarPago(usuario, m.getId());
    }
